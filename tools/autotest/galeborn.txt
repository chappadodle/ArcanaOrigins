# Galeborn (see the design spec): its three spells on the skill keys at a slowed 200-health husk
# (shots; the husk's health logged after each), the double jump (the player's height logged at the
# top of a plain jump and of a double jump), no fall damage, and 3 hearts less.
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule mobGriefing false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd tp @s 0.5 100 0.5 -90 0
wait 40
cmd kill @e[type=!player]
cmd fill -8 99 -12 30 99 12 minecraft:stone
cmd fill -8 100 -12 30 110 12 minecraft:air
cmd origin set @s neoorigins:origin arcanaorigins:galeborn
wait 20
screen close
cmd summon husk 9.5 100 0.5 {PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd effect give @e[type=husk] minecraft:slowness 200 255 true
cmd effect give @e[type=husk] minecraft:weakness 200 255 true
wait 10
cmd tp @s 0.5 100 0.5 -90 3
wait 5
# 1. Wind Blade.
skill 1
wait 3
shot wind_blade_flight
wait 15
cmd data get entity @e[type=husk,limit=1] Health
shot wind_blade_hit
# 2. Gale Dash, from behind (toward the husk; it ends short of it).
cmd tp @e[type=husk] 14.5 100 0.5
cmd tp @s 0.5 100 0.5 -90 0
wait 5
camera back
skill 2
wait 3
shot gale_dash
wait 15
cmd data get entity @s Pos
camera first
# 3. Stormeye on the husk.
cmd tp @e[type=husk] 9.5 100 0.5
cmd tp @s 0.5 100 0.5 -90 8
wait 5
skill 3
wait 30
shot stormeye
wait 70
cmd data get entity @e[type=husk,limit=1] Health
# 4. Updraft: a plain jump, then a jump in mid-air (the top of each logged).
cmd kill @e[type=husk]
cmd tp @s 0.5 100 0.5 -90 0
wait 40
key jump down
wait 2
key jump up
wait 4
cmd data get entity @s Pos[1]
wait 30
key jump down
wait 2
key jump up
wait 4
key jump down
wait 2
key jump up
wait 6
cmd data get entity @s Pos[1]
shot double_jump
wait 40
# 5. Featherfall: a 20-block drop; health logged after.
cmd effect give @s minecraft:instant_health 1 10 true
wait 5
cmd tp @s 0.5 120 0.5 -90 0
wait 60
cmd data get entity @s Health
cmd attribute @s minecraft:generic.max_health get
cmd kill @e[type=!player]
wait 10
quit
