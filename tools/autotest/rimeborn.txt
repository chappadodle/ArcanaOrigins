# Rimeborn (see the design spec): its three spells on the skill keys at a slowed 200-health husk
# (shots; the husk's health logged after each), the frost shield seen from behind, and Frost Walker:
# standing at a pool's edge, the water by the player freezes (checked).
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule mobGriefing false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd tp @s 0.5 100 0.5 -90 0
wait 40
cmd kill @e[type=!player]
cmd fill -8 99 -12 30 99 12 minecraft:stone
cmd fill -8 100 -12 30 110 12 minecraft:air
cmd origin set @s neoorigins:origin arcanaorigins:rimeborn
wait 20
screen close
cmd summon husk 9.5 100 0.5 {PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd effect give @e[type=husk] minecraft:slowness 200 255 true
cmd effect give @e[type=husk] minecraft:weakness 200 255 true
wait 10
cmd tp @s 0.5 100 0.5 -90 3
wait 5
# 1. Icicle.
skill 1
wait 4
shot icicle_flight
wait 20
cmd data get entity @e[type=husk,limit=1] Health
shot icicle_hit
# 2. Frost Shield, from behind.
skill 2
wait 10
camera back
wait 5
shot frost_shield
camera first
wait 30
# 3. Frost Nova, with the husk close.
cmd tp @e[type=husk] 3.5 100 0.5
wait 5
skill 3
wait 8
shot frost_nova
wait 20
cmd data get entity @e[type=husk,limit=1] Health
cmd data get entity @e[type=husk,limit=1] active_effects
# 4. Frost Walker: a pool beside the player.
cmd kill @e[type=husk]
cmd fill 2 99 -2 6 99 2 minecraft:water
cmd tp @s 1.5 100 0.5 -90 40
wait 30
cmd execute if block 2 99 0 minecraft:frosted_ice
shot frost_walk
cmd fill 2 99 -2 6 99 2 minecraft:stone
cmd kill @e[type=!player]
wait 10
quit
